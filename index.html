import { Scene, Voxel } from 'voxel-core';

const scene = new Scene();
const cube = new Voxel({
    size: [2, 2, 2],
    color: '#00ffcc',
    position: [0, 5, 0]
});

scene.add(cube);
scene.animate(() => {
    cube.rotation.y += 0.01;
});
